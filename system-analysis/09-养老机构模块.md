# 养老机构模块分析

## 1. 模块概述

养老机构模块提供养老机构信息管理、地图展示、预约参观、时光传记等功能，是连接老年人与养老服务的桥梁模块。

## 4. 数据模型

### 4.1 核心实体

```python
class CareFacility(Base):
    """养老机构表"""
    __tablename__ = "care_facilities"
    
    id = Column(Integer, primary_key=True)
    name = Column(String(200), nullable=False)
    description = Column(Text)
    address = Column(String(500))
    latitude = Column(Float)
    longitude = Column(Float)
    
    # 联系方式
    phone = Column(String(20))
    website = Column(String(200))
    
    # 服务信息
    facility_type = Column(String(50))  # 养老院、护理院、日间照料等
    capacity = Column(Integer)
    available_beds = Column(Integer)
    price_range = Column(String(50))
    services = Column(JSON)  # 服务项目列表
    
    # 评分
    rating = Column(Float, default=0)
    review_count = Column(Integer, default=0)
    
    # 状态
    status = Column(String(20), default="active")
    created_at = Column(DateTime, default=datetime.utcnow)

class Reservation(Base):
    """预约参观表"""
    __tablename__ = "reservations"
    
    id = Column(Integer, primary_key=True)
    facility_id = Column(Integer, ForeignKey("care_facilities.id"))
    user_id = Column(Integer, ForeignKey("users.id"))
    
    # 预约信息
    reservation_date = Column(Date, nullable=False)
    reservation_time = Column(String(20))
    visitor_count = Column(Integer, default=1)
    contact_name = Column(String(50))
    contact_phone = Column(String(20))
    
    # 状态
    status = Column(String(20), default="pending")  # pending, confirmed, completed, cancelled
    notes = Column(Text)
    
    created_at = Column(DateTime, default=datetime.utcnow)
    reviewed_at = Column(DateTime)
    reviewed_by = Column(Integer, ForeignKey("users.id"))

class Biography(Base):
    """时光传记表"""
    __tablename__ = "biographies"
    
    id = Column(Integer, primary_key=True)
    patient_id = Column(Integer, ForeignKey("patients.id"))
    title = Column(String(200), nullable=False)
    content = Column(Text)
    
    # 多媒体
    images = Column(JSON)
    videos = Column(JSON)
    audio = Column(JSON)
    
    # 分类
    category = Column(String(50))  # 人生故事、家族历史、重要时刻等
    tags = Column(JSON)
    
    # 状态
    status = Column(String(20), default="draft")
    created_by = Column(Integer, ForeignKey("users.id"))
    created_at = Column(DateTime, default=datetime.utcnow)
```
