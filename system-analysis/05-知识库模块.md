# 知识库模块分析

## 1. 模块概述

知识库模块提供知识文档管理、知识空间、审核流程、版本控制等功能，支持知识问答和知识统计分析。模块采用多空间设计，支持知识的分类管理和权限控制。

```python
class KnowledgeSpace(Base):
    """知识空间表"""
    __tablename__ = "knowledge_spaces"
    
    id = Column(Integer, primary_key=True, index=True)
    name = Column(String(100), nullable=False)
    description = Column(Text)
    category = Column(String(50))
    is_public = Column(Boolean, default=True)
    owner_id = Column(Integer, ForeignKey("users.id"))
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
    
    # 关系
    documents = relationship("KnowledgeDocument", back_populates="space")

class KnowledgeDocument(Base):
    """知识文档表"""
    __tablename__ = "knowledge_documents"
    
    id = Column(Integer, primary_key=True, index=True)
    space_id = Column(Integer, ForeignKey("knowledge_spaces.id"))
    title = Column(String(200), nullable=False)
    content = Column(Text, nullable=False)
    summary = Column(Text)
    category = Column(String(50))
    tags = Column(JSON)
    author_id = Column(Integer, ForeignKey("users.id"))
    status = Column(String(20), default="draft")  # draft, pending, published, rejected
    version = Column(Integer, default=1)
    view_count = Column(Integer, default=0)
    like_count = Column(Integer, default=0)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
    published_at = Column(DateTime)
    
    # 关系
    space = relationship("KnowledgeSpace", back_populates="documents")
    versions = relationship("DocumentVersion", back_populates="document")
    reviews = relationship("KnowledgeReview", back_populates="document")

class DocumentVersion(Base):
    """文档版本表"""
    __tablename__ = "document_versions"
    
    id = Column(Integer, primary_key=True, index=True)
    document_id = Column(Integer, ForeignKey("knowledge_documents.id"))
    version_number = Column(Integer, nullable=False)
    content = Column(Text, nullable=False)
    change_summary = Column(Text)
    created_by = Column(Integer, ForeignKey("users.id"))
    created_at = Column(DateTime, default=datetime.utcnow)
    
    # 关系
    document = relationship("KnowledgeDocument", back_populates="versions")

class KnowledgeReview(Base):
    """知识审核表"""
    __tablename__ = "knowledge_reviews"
    
    id = Column(Integer, primary_key=True, index=True)
    document_id = Column(Integer, ForeignKey("knowledge_documents.id"))
    reviewer_id = Column(Integer, ForeignKey("users.id"))
    status = Column(String(20), default="pending")  # pending, approved, rejected
    comment = Column(Text)
    reviewed_at = Column(DateTime)
    created_at = Column(DateTime, default=datetime.utcnow)
    
    # 关系
    document = relationship("KnowledgeDocument", back_populates="reviews")
```

```python
# 建议集成Elasticsearch
from elasticsearch import AsyncElasticsearch

class ElasticsearchRetriever:
    """Elasticsearch检索器"""
    
    def __init__(self, hosts: list):
        self.client = AsyncElasticsearch(hosts)
    
    async def index_document(self, doc: Document):
        """索引文档"""
        await self.client.index(
            index="knowledge",
            id=doc.id,
            document={
                "title": doc.title,
                "content": doc.content,
                "summary": doc.summary,
                "tags": doc.tags,
                "category": doc.category,
                "space_id": doc.space_id,
                "created_at": doc.created_at
            }
        )
    
    async def search(self, query: str, space_id: int = None, 
                     limit: int = 10) -> List[dict]:
        """全文搜索"""
        must = [
            {
                "multi_match": {
                    "query": query,
                    "fields": ["title^2", "content", "summary", "tags"],
                    "type": "best_fields"
                }
            }
        ]
        
        if space_id:
            must.append({"term": {"space_id": space_id}})
        
        response = await self.client.search(
            index="knowledge",
            query={"bool": {"must": must}},
            size=limit,
            highlight={
                "fields": {
                    "content": {"fragment_size": 200, "number_of_fragments": 3}
                }
            }
        )
        
        return response["hits"]["hits"]
```

```python
# 建议增加知识关联关系
class KnowledgeRelation(Base):
    """知识关联表"""
    __tablename__ = "knowledge_relations"
    
    id = Column(Integer, primary_key=True)
    source_id = Column(Integer, ForeignKey("knowledge_documents.id"))
    target_id = Column(Integer, ForeignKey("knowledge_documents.id"))
    relation_type = Column(String(50))  # related_to, prerequisite, extends, etc.
    weight = Column(Float, default=1.0)
    created_at = Column(DateTime, default=datetime.utcnow)

class KnowledgeGraphService:
    """知识图谱服务"""
    
    def __init__(self, db: Session):
        self.db = db
    
    async def get_related_documents(self, doc_id: int, 
                                     relation_type: str = None) -> List[Document]:
        """获取关联文档"""
        query = self.db.query(KnowledgeRelation).filter(
            KnowledgeRelation.source_id == doc_id
        )
        if relation_type:
            query = query.filter(KnowledgeRelation.relation_type == relation_type)
        
        relations = query.all()
        doc_ids = [r.target_id for r in relations]
        
        return self.db.query(Document).filter(Document.id.in_(doc_ids)).all()
    
    async def create_relation(self, source_id: int, target_id: int, 
                               relation_type: str, weight: float = 1.0):
        """创建关联关系"""
        relation = KnowledgeRelation(
            source_id=source_id,
            target_id=target_id,
            relation_type=relation_type,
            weight=weight
        )
        self.db.add(relation)
        self.db.commit()
```

```python
class SpacePermission(Base):
    """空间权限表"""
    __tablename__ = "space_permissions"
    
    id = Column(Integer, primary_key=True)
    space_id = Column(Integer, ForeignKey("knowledge_spaces.id"))
    user_id = Column(Integer, ForeignKey("users.id"), nullable=True)
    role_id = Column(Integer, ForeignKey("roles.id"), nullable=True)
    permission_type = Column(String(20))  # read, write, admin
    created_at = Column(DateTime, default=datetime.utcnow)

class SpacePermissionService:
    """空间权限服务"""
    
    def check_permission(self, user_id: int, space_id: int, 
                          permission_type: str) -> bool:
        """检查用户对空间的权限"""
        # 检查用户直接权限
        user_perm = self.db.query(SpacePermission).filter(
            SpacePermission.user_id == user_id,
            SpacePermission.space_id == space_id,
            SpacePermission.permission_type == permission_type
        ).first()
        
        if user_perm:
            return True
        
        # 检查角色权限
        user_roles = self.db.query(UserRoleAssignment).filter(
            UserRoleAssignment.user_id == user_id
        ).all()
        
        role_ids = [r.role_id for r in user_roles]
        
        role_perm = self.db.query(SpacePermission).filter(
            SpacePermission.role_id.in_(role_ids),
            SpacePermission.space_id == space_id,
            SpacePermission.permission_type == permission_type
        ).first()
        
        return role_perm is not None
```
