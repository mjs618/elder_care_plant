## 4. 数据模型

### 4.1 核心实体

```python
class Patient(Base):
    """患者表"""
    __tablename__ = "patients"
    
    id = Column(Integer, primary_key=True, index=True)
    name = Column(String(100), nullable=False)
    gender = Column(String(10), nullable=False)
    birth_date = Column(Date, nullable=False)
    id_card = Column(String(20))
    phone = Column(String(20))
    address = Column(String(500))
    emergency_contact = Column(String(50))
    emergency_phone = Column(String(20))
    
    # 健康档案
    blood_type = Column(String(5))
    allergies = Column(JSON)  # 过敏史
    medical_history = Column(JSON)  # 病史
    family_history = Column(JSON)  # 家族史
    
    # 照护信息
    care_level = Column(String(20))  # 照护等级
    caregiver_id = Column(Integer, ForeignKey("users.id"))
    
    # 状态
    status = Column(String(20), default="active")  # active, discharged, deceased
    
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
    
    # 关系
    caregiver = relationship("User", back_populates="patients")
    health_data = relationship("HealthData", back_populates="patient")
    assessments = relationship("Assessment", back_populates="patient")
    medications = relationship("PatientMedication", back_populates="patient")
```

```python
from cryptography.fernet import Fernet

class Patient(Base):
    _id_card = Column("id_card", String(100))
    
    @property
    def id_card(self):
        if self._id_card:
            return decrypt(self._id_card)
        return None
    
    @id_card.setter
    def id_card(self, value):
        if value:
            self._id_card = encrypt(value)
        else:
            self._id_card = None
```

### 7.2 数据校验

```python
# 建议增加数据校验
from pydantic import BaseModel, validator
import re

class PatientCreate(BaseModel):
    name: str
    id_card: Optional[str]
    phone: Optional[str]
    
    @validator('id_card')
    def validate_id_card(cls, v):
        if v and not re.match(r'^\d{17}[\dXx]$', v):
            raise ValueError('身份证号格式不正确')
        return v
    
    @validator('phone')
    def validate_phone(cls, v):
        if v and not re.match(r'^1\d{10}$', v):
            raise ValueError('手机号格式不正确')
        return v
```
