# AI智能模块分析

## 1. 模块概述

AI智能模块是系统的核心创新功能，集成多种大语言模型（ChatGLM、Qwen、Baichuan等），提供智能健康问答、症状分析、个性化建议等功能。模块采用延迟加载策略，支持模型热切换和内存管理。

```python
class ModelManager:
    """模型管理器 - 延迟加载模式"""
    
    def __init__(self):
        self.models: Dict[str, LocalAIModel] = {}
        self.current_model: Optional[str] = None

        self.vector_index = None
        self._initialized = False
        self.model_size_cache: Dict[str, float] = {}
        self.model_memory_cache: Dict[str, int] = {}
        self.max_concurrent_models = 2
    
    def initialize(self) -> bool:
        """手动初始化模型（延迟加载）"""
        if self._initialized:
            return True
        
        try:
            self._load_models()
            self._load_embedding_model()
            self._initialized = True
            return True
        except Exception as e:
            logger.error(f"AI模型初始化失败: {e}")
            return False
    
    def load_model(self, model_name: str) -> bool:
        """加载单个模型到内存"""
        if model_name in self.models:
            return True
        
        if len(self.models) >= self.max_concurrent_models:
            logger.warning(f"已达到最大并发模型数 {self.max_concurrent_models}")
            return False
        
        # 加载模型逻辑...
    
    def unload_model(self, model_name: str) -> bool:
        """从内存卸载单个模型"""
        if model_name not in self.models:
            return False
        
        del self.models[model_name]
        
        if self.current_model == model_name:
            if self.models:
                self.current_model = list(self.models.keys())[0]
            else:
                self.current_model = None
        
        if torch.cuda.is_available():
            torch.cuda.empty_cache()
        
        return True
```

### 4.3 ChatGLM模型实现

```python
class ChatGLMModel(LocalAIModel):
    """ChatGLM模型"""
    
    def generate_response(self, prompt: str, context: str = "") -> str:
        """ChatGLM专用生成方法"""
        try:
            full_prompt = self._build_prompt(prompt, context)
            
            inputs = self.tokenizer(
                full_prompt, 
                return_tensors="pt", 
                max_length=1024, 
                truncation=True,
                padding=True
            )
            
            if "attention_mask" not in inputs:
                inputs["attention_mask"] = torch.ones_like(inputs["input_ids"])
            
            if self.device == "cuda":
                inputs = {k: v.cuda() for k, v in inputs.items()}
            
            with torch.no_grad():
                max_tokens = 128 if self.device == "cpu" else 512
                outputs = self.model.generate(
                    input_ids=inputs["input_ids"],
                    attention_mask=inputs.get("attention_mask"),
                    max_new_tokens=max_tokens,
                    do_sample=True,
                    temperature=self.config.temperature,
                    top_p=self.config.top_p,
                    num_beams=1,
                    pad_token_id=self.tokenizer.pad_token_id or self.tokenizer.eos_token_id,
                    use_cache=False
                )
            
            response = self.tokenizer.decode(outputs[0], skip_special_tokens=True)
            return self._clean_response(response, full_prompt)
            
        except Exception as e:
            logger.error(f"ChatGLM生成回复失败: {e}")
            return "抱歉，我现在无法回答您的问题，请稍后再试。"
```

## 6. 评估分析

### 6.1 优点

1. **多模型支持**: 支持ChatGLM、Qwen、Baichuan等多种国产大模型
2. **延迟加载**: 模型按需加载，减少启动时间和内存占用
3. **内存管理**: 支持模型热切换和并发限制
4. **流式响应**: 支持SSE流式输出，提升用户体验
5. **向量增强**: 集成FAISS向量检索，增强问答质量
6. **OCR集成**: 支持图片文字识别，扩展应用场景

### 6.2 问题与风险

| 问题 | 严重程度 | 描述 |
|------|----------|------|
| 模型兼容性 | 高 | 不同模型API差异大，维护成本高 |
| 内存占用 | 高 | 大模型内存占用大，可能OOM |
| CPU推理慢 | 中 | CPU模式下推理速度慢 |
| 错误处理 | 中 | 部分异常处理不够完善 |
| 缓存缺失 | 低 | 相似问题缺少缓存机制 |

### 7.2 增加问答缓存

```python
# 建议增加问答缓存机制
import hashlib
from functools import lru_cache

class CachedQAService:
    def __init__(self, model_manager: ModelManager, cache_ttl: int = 3600):
        self.model_manager = model_manager
        self.cache_ttl = cache_ttl
        self._cache = {}
    
    def _get_cache_key(self, question: str, context: str = "") -> str:
        content = f"{question}:{context}"
        return hashlib.md5(content.encode()).hexdigest()
    
    async def get_answer(self, question: str, context: str = "") -> str:
        cache_key = self._get_cache_key(question, context)
        
        # 检查缓存
        if cache_key in self._cache:
            cached = self._cache[cache_key]
            if time.time() - cached['timestamp'] < self.cache_ttl:
                return cached['answer']
        
        # 调用模型
        answer = await self.model_manager.generate_response(question, context)
        
        # 存入缓存
        self._cache[cache_key] = {
            'answer': answer,
            'timestamp': time.time()
        }
        
        return answer
```

### 7.4 完善错误处理

```python
# 建议增加统一的错误处理
from enum import Enum
from dataclasses import dataclass

class AIErrorType(Enum):
    MODEL_NOT_LOADED = "model_not_loaded"
    MODEL_LOAD_FAILED = "model_load_failed"
    GENERATION_FAILED = "generation_failed"
    TIMEOUT = "timeout"
    RATE_LIMITED = "rate_limited"

@dataclass
class AIError:
    error_type: AIErrorType
    message: str
    retry_after: int = 0
    fallback_response: str = ""

class AIServiceException(Exception):
    def __init__(self, error: AIError):
        self.error = error
        super().__init__(error.message)

def handle_ai_error(func):
    @wraps(func)
    async def wrapper(*args, **kwargs):
        try:
            return await func(*args, **kwargs)
        except torch.cuda.OutOfMemoryError:
            raise AIServiceException(AIError(
                error_type=AIErrorType.MODEL_LOAD_FAILED,
                message="GPU内存不足，请卸载其他模型后重试",
                fallback_response="抱歉，系统资源不足，请稍后再试。"
            ))
        except asyncio.TimeoutError:
            raise AIServiceException(AIError(
                error_type=AIErrorType.TIMEOUT,
                message="模型响应超时",
                fallback_response="抱歉，响应时间过长，请简化问题后重试。"
            ))
    return wrapper
```
