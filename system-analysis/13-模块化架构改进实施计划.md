```python
"""
模块契约基类
定义模块间通信的标准化接口规范
"""
from pydantic import BaseModel, Field
from typing import Optional, ClassVar
from datetime import datetime
from enum import Enum


class ContractVersion(str, Enum):
    """契约版本枚举"""
    V1 = "1.0"
    V2 = "2.0"


class BaseContract(BaseModel):
    """
    契约基类
    所有模块对外接口都应继承此类
    """
    
    # 契约版本 - 子类必须定义
    contract_version: ClassVar[str] = ContractVersion.V1.value
    
    # 资源ID
    id: str = Field(..., description="资源唯一标识")
    
    # 租户ID（多租户场景）
    tenant_id: Optional[str] = Field(None, description="租户ID")
    
    # 创建和更新时间
    created_at: Optional[datetime] = Field(None, description="创建时间")
    updated_at: Optional[datetime] = Field(None, description="更新时间")
    
    class Config:
        # 允许额外字段（向前兼容）
        extra = "allow"
        # 使用枚举值
        use_enum_values = True


class ContractClient:
    """
    契约客户端基类
    用于模块间HTTP调用
    """
    
    def __init__(self, base_url: str, module_name: str):
        self.base_url = base_url.rstrip("/")
        self.module_name = module_name
    
    async def _request(
        self, 
        method: str, 
        path: str, 
        **kwargs
    ) -> dict:
        """
        发送HTTP请求
        """
        import httpx
        
        url = f"{self.base_url}{path}"
        
        async with httpx.AsyncClient() as client:
            response = await client.request(
                method, 
                url, 
                timeout=30.0,
                **kwargs
            )
            response.raise_for_status()
            return response.json()
    
    async def get(self, path: str, **kwargs) -> dict:
        return await self._request("GET", path, **kwargs)
    
    async def post(self, path: str, **kwargs) -> dict:
        return await self._request("POST", path, **kwargs)
    
    async def put(self, path: str, **kwargs) -> dict:
        return await self._request("PUT", path, **kwargs)
    
    async def delete(self, path: str, **kwargs) -> dict:
        return await self._request("DELETE", path, **kwargs)
```

```python
                "room_number": "A区101",
                "bed_number": "01",
                "emergency_contact": "张四（子）",

            }
        }

    """
    患者列表契约

    total: int = Field(0, description="总数")

    患者模块客户端

        super().__init__(base_url, "patient_mgmt")

        获取单个患者信息

        return PatientContract(**response.get("data", {}))

        self, 
        page: int = 1, 
        size: int = 20,
        search: Optional[str] = None
    ) -> PatientListContract:
        """
        获取患者列表
        """
        params = {"page": page, "size": size}
        if search:
            params["search"] = search
        
        response = await self.get("/api/v1/patients", params=params)
        data = response.get("data", {})
        
        return PatientListContract(
            items=[PatientContract(**item) for item in data.get("items", [])],
            total=data.get("total", 0),
            page=data.get("page", 1),
            size=data.get("size", 20)
        )
    
    async def check_patient_exists(self, patient_id: str) -> bool:
        """
        检查患者是否存在
        """
        try:
            await self.get_patient(patient_id)
            return True
        except Exception:
            return False
```

```python
"""
模块事件总线
基于 RabbitMQ 实现模块间异步通信
"""
import json
import asyncio
from typing import Callable, Any, Optional, Dict, List
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
import structlog

import aio_pika
from aio_pika import Message, DeliveryMode
from aio_pika.abc import AbstractIncomingMessage

logger = structlog.get_logger()


class EventType(str, Enum):
    """事件类型枚举"""
    # 患者相关
    PATIENT_CREATED = "patient.created"
    PATIENT_UPDATED = "patient.updated"
    PATIENT_DELETED = "patient.deleted"
    
    # 评估相关
    ASSESSMENT_CREATED = "assessment.created"
    ASSESSMENT_UPDATED = "assessment.updated"
    ASSESSMENT_COMPLETED = "assessment.completed"
    
    # 健康相关
    HEALTH_ALERT = "health.alert"
    VITAL_SIGN_RECORDED = "health.vital_sign_recorded"
    
    # AI相关
    AI_CHAT_COMPLETED = "ai.chat_completed"


@dataclass
class Event:
    """
    事件数据结构
    """
    event_type: str                    # 事件类型
    source_module: str                 # 来源模块
    payload: Dict[str, Any]            # 事件载荷
    event_id: str = field(default_factory=lambda: str(__import__('uuid').uuid4()))
    timestamp: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    version: str = "1.0"               # 事件版本
    
    def to_dict(self) -> dict:
        return {
            "event_id": self.event_id,
            "event_type": self.event_type,
            "source_module": self.source_module,
            "payload": self.payload,
            "timestamp": self.timestamp,
            "version": self.version,
        }
    
    @classmethod
    def from_dict(cls, data: dict) -> "Event":
        return cls(
            event_id=data["event_id"],
            event_type=data["event_type"],
            source_module=data["source_module"],
            payload=data["payload"],
            timestamp=data["timestamp"],
            version=data["version"],
        )


class EventBus:
    """
    事件总线
    提供发布/订阅模式的事件通信
    """
    
    EXCHANGE_NAME = "elder_care_events"
    
    def __init__(self, rabbitmq_url: str):
        self.url = rabbitmq_url
        self.connection: Optional[aio_pika.RobustConnection] = None
        self.channel: Optional[aio_pika.Channel] = None
        self.exchange: Optional[aio_pika.Exchange] = None
        self._subscribers: Dict[str, List[Callable]] = {}
        self._queues: Dict[str, aio_pika.Queue] = {}
    
    async def connect(self) -> None:
        """
        建立连接
        """
        logger.info("event_bus_connecting", url=self.url)
        
        self.connection = await aio_pika.connect_robust(self.url)
        self.channel = await self.connection.channel()
        
        # 声明主题交换机
        self.exchange = await self.channel.declare_exchange(
            self.EXCHANGE_NAME,
            aio_pika.ExchangeType.TOPIC,
            durable=True
        )
        
        logger.info("event_bus_connected")
    
    async def disconnect(self) -> None:
        """
        断开连接
        """
        if self.connection:
            await self.connection.close()
            logger.info("event_bus_disconnected")
    
    async def publish(self, event: Event) -> None:
        """
        发布事件
        """
        if not self.exchange:
            raise RuntimeError("Event bus not connected")
        
        message = Message(
            body=json.dumps(event.to_dict()).encode(),
            content_type="application/json",
            delivery_mode=DeliveryMode.PERSISTENT,
            message_id=event.event_id,
            timestamp=datetime.utcnow(),
        )
        
        routing_key = f"{event.source_module}.{event.event_type}"
        
        await self.exchange.publish(
            message,
            routing_key=routing_key
        )
        
        logger.info(
            "event_published",
            event_id=event.event_id,
            event_type=event.event_type,
            routing_key=routing_key
        )
    
    async def subscribe(
        self,
        module_name: str,
        event_pattern: str,
        handler: Callable[[Event], Any]
    ) -> None:
        """
        订阅事件
        
        Args:
            module_name: 订阅模块名称
            event_pattern: 事件模式，支持通配符
                          例如: "patient.*" 订阅所有患者事件
                               "assessment.completed" 订阅特定事件
            handler: 事件处理函数
        """
        if not self.channel or not self.exchange:
            raise RuntimeError("Event bus not connected")
        
        # 创建模块专属队列
        queue_name = f"{module_name}.{event_pattern.replace('*', 'all').replace('.', '_')}"
        
        if queue_name not in self._queues:
            queue = await self.channel.declare_queue(
                queue_name,
                durable=True,
                arguments={
                    "x-message-ttl": 86400000,  # 消息存活时间: 24小时
                }
            )
            
            # 绑定队列到交换机
            await queue.bind(self.exchange, routing_key=event_pattern)
            self._queues[queue_name] = queue
            
            logger.info(
                "event_queue_created",
                queue_name=queue_name,
                event_pattern=event_pattern
            )
        
        # 注册处理函数
        if event_pattern not in self._subscribers:
            self._subscribers[event_pattern] = []
        self._subscribers[event_pattern].append(handler)
        
        # 开始消费
        async def process_message(message: AbstractIncomingMessage):
            async with message.process():
                try:
                    data = json.loads(message.body.decode())
                    event = Event.from_dict(data)
                    
                    logger.info(
                        "event_received",
                        event_id=event.event_id,
                        event_type=event.event_type
                    )
                    
                    for h in self._subscribers.get(event_pattern, []):
                        try:
                            result = h(event)
                            if asyncio.iscoroutine(result):
                                await result
                        except Exception as e:
                            logger.error(
                                "event_handler_error",
                                event_id=event.event_id,
                                error=str(e)
                            )
                except Exception as e:
                    logger.error("event_process_error", error=str(e))
        
        await self._queues[queue_name].consume(process_message)
        
        logger.info(
            "event_subscribed",
            module=module_name,
            pattern=event_pattern
        )


# 全局事件总线实例
_event_bus: Optional[EventBus] = None


def get_event_bus() -> EventBus:
    """获取全局事件总线实例"""
    global _event_bus
    if _event_bus is None:
        raise RuntimeError("Event bus not initialized")
    return _event_bus


async def init_event_bus(rabbitmq_url: str) -> EventBus:
    """初始化全局事件总线"""
    global _event_bus
    _event_bus = EventBus(rabbitmq_url)
    await _event_bus.connect()
    return _event_bus
```

```python
"""
患者管理模块 - API路由
"""
from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy import select, or_, desc, func
from sqlalchemy.ext.asyncio import AsyncSession
import uuid

from shared.database import get_db
from shared.dependencies import get_current_user, require_module
from contracts.patient_contract import PatientContract, PatientListContract
from modules.patient_mgmt.models.patient import Patient
from modules.patient_mgmt.schemas.patient import PatientCreate, PatientUpdate
from shared.event_bus import Event, get_event_bus, EventType

router = APIRouter()
depends_module = Depends(require_module("patient_mgmt"))


@router.get("", response_model=dict, dependencies=[depends_module])
async def list_patients(
    search: str | None = None,
    page: int = Query(1, ge=1),
    size: int = Query(20, ge=1, le=100),
    db: AsyncSession = Depends(get_db),
    current_user = Depends(get_current_user),
):
    """获取患者列表"""
    query = select(Patient).where(
        Patient.is_deleted == False
    ).order_by(desc(Patient.created_at))
    
    if search:
        query = query.where(
            or_(
                Patient.full_name.ilike(f"%{search}%"),
                Patient.id_card_num.ilike(f"%{search}%"),
                Patient.room_number.ilike(f"%{search}%"),
            )
        )
    
    offset = (page - 1) * size
    paginated_query = query.offset(offset).limit(size)
    
    result = await db.execute(paginated_query)
    patients = result.scalars().all()
    
    count_query = select(func.count()).select_from(query.subquery())
    total = await db.scalar(count_query) or 0
    
    return {
        "code": 200,
        "message": "success",
        "data": {
            "items": [PatientContract.model_validate(p).model_dump() for p in patients],
            "total": total,
            "page": page,
            "size": size,
        }
    }


@router.post("", response_model=dict, dependencies=[depends_module])
```

#### 1.3.4 迁移脚本

```python
"""
模块迁移脚本
将现有代码迁移到新的模块目录结构
"""
import os
import shutil
from pathlib import Path

# 项目根目录
ROOT_DIR = Path(__file__).parent.parent

# 模块映射
MODULE_MIGRATION = {
    "patient_mgmt": {
        "api": ["patients.py"],
        "models": ["patient.py"],
        "schemas": ["patient.py"],
    },
    "assessment": {
        "api": ["assessments.py"],
        "models": ["assessment.py"],
        "schemas": ["assessment.py"],
    },
}


def create_module_structure(module_name: str):
    """创建模块目录结构"""
    module_dir = ROOT_DIR / "modules" / module_name
    
    dirs = [
        module_dir,
        module_dir / "api",
        module_dir / "models",
        module_dir / "schemas",
        module_dir / "services",
        module_dir / "subscribers",
    ]
    
    for d in dirs:
        d.mkdir(parents=True, exist_ok=True)
        (d / "__init__.py").touch()
    
    print(f"Created module structure: {module_name}")


def migrate_files(module_name: str, file_mapping: dict):
    """迁移文件"""
    module_dir = ROOT_DIR / "modules" / module_name
    
    for target_dir, files in file_mapping.items():
        for file_name in files:
            src = ROOT_DIR / "app" / target_dir / file_name
            dst = module_dir / target_dir / file_name
            
            if src.exists():
                shutil.copy2(src, dst)
                print(f"Migrated: {src} -> {dst}")


def main():
    """执行迁移"""
    print("Starting module migration...")
    
    # 创建共享目录
    (ROOT_DIR / "shared").mkdir(exist_ok=True)
    (ROOT_DIR / "contracts").mkdir(exist_ok=True)
    (ROOT_DIR / "modules").mkdir(exist_ok=True)
    
    # 迁移各模块
    for module_name, file_mapping in MODULE_MIGRATION.items():
        create_module_structure(module_name)
        migrate_files(module_name, file_mapping)
    
    print("\nMigration completed!")
    print("Please review the migrated files and update imports.")


if __name__ == "__main__":
    main()
```

```sql
-- 创建模块级 Schema
CREATE SCHEMA IF NOT EXISTS core;        -- 核心数据

    LIKE public.patients INCLUDING ALL
```

```yaml
name: Patient Module CI/CD

on:
  push:
    paths:
      - 'backend/modules/patient_mgmt/**'

      - '.github/workflows/patient-module.yml'

      - uses: actions/checkout@v4
      
      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'

        run: |
          cd backend
          pip install poetry
          poetry install
      
      - name: Run tests
        run: |
          cd backend
          poetry run pytest modules/patient_mgmt/tests -v

  build:
    needs: test
    runs-on: ubuntu-latest
    outputs:
      image_tag: ${{ steps.meta.outputs.tags }}
    
    steps:
      - uses: actions/checkout@v4
      
      - name: Log in to Container Registry
        uses: docker/login-action@v3

          username: ${{ github.actor }}

          tags: ${{ steps.meta.outputs.tags }}
          labels: ${{ steps.meta.outputs.labels }}

    runs-on: ubuntu-latest
    if: github.ref == 'refs/heads/main'
```

#### 3.1.3 动态加载远程模块

```typescript
// frontend/src/router/dynamicRoutes.ts
import { defineAsyncComponent } from 'vue'

const remoteModules = {
  assessment: {
    url: 'http://localhost:8002/assets/remoteEntry.js',
    exposes: {
      AssessmentList: './AssessmentList',
      AssessmentForm: './AssessmentForm',
    },
  },
  patient: {
    url: 'http://localhost:8001/assets/remoteEntry.js',
    exposes: {
      PatientList: './PatientList',
      PatientForm: './PatientForm',
    },
  },
}

export async function loadRemoteModule(
  moduleName: string,
  componentName: string
) {
  const module = remoteModules[moduleName]
  if (!module) {
    throw new Error(`Module ${moduleName} not found`)
  }

  // 动态导入远程组件
  const remoteEntry = await import(module.url)
  const component = await remoteEntry.get(componentName)
  
  return defineAsyncComponent(() => component())
}

// 在路由中使用
export function createDynamicRoutes(activeModules: string[]) {
  const routes = []
  
  if (activeModules.includes('assessment')) {
    routes.push({
      path: '/assessments',
      component: () => loadRemoteModule('assessment', 'AssessmentList'),
    })
  }
  
  if (activeModules.includes('patient_mgmt')) {
    routes.push({
      path: '/patients',
      component: () => loadRemoteModule('patient', 'PatientList'),
    })
  }
  
  return routes
}
```
